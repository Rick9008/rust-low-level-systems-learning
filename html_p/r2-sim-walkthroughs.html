Code 走讀——cancel 與 on_done 的收尾協定

// cancel:停止再派;在飛的塊等自然 done。不認識的 id 是 no-op——cancel 天生要冪等。
pub fn cancel(&mut self, id: u64) {
    let Some(st) = self.reqs.get_mut(&id) else { return; };  // 冪等:已完成/沒來過 → no-op
    st.cancelled = true;
    st.next_block = st.blocks_total;          // 這單從此派不出塊(停派)
    if st.in_flight == 0 {
        self.reqs.remove(&id);                 // 沒有在飛的塊,現在就能退場
    }
}

pub fn on_done(&mut self, bus: &mut impl DmaBus, eid: u32) {
    let (rid, _b) = self.owner[eid as usize].take()   // take:拿走路由,engine 才算空
        .expect("done 來自一台沒派工的 engine");
    self.free.push_back(eid);                  // 先還台——engine 回收與單的成敗無關
    let st = self.reqs.get_mut(&rid).unwrap();
    st.in_flight -= 1;
    if st.cancelled {
        if st.in_flight == 0 { self.reqs.remove(&rid); }  // 最後一塊回來 → 靜默退場
        return;                           // 不計數、不 submit——cancel 三律的後兩律
    }
    st.done += 1;
    if st.done == st.blocks_total {            // 完成判定唯一正解:計數歸零
        bus.submit_dma_request_result_done(rid);
        self.reqs.remove(&rid);
    }
}
dispatch 的一個攤銷細節

待派 queue 前端可能積著「派完了/取消了」的殭屍單,dispatch 用一個內層 loop 順手 pop_front 清掉再派。每張單一生最多被 pop 一次 → 攤銷 O(1), 講 Big-O 時說得出「攤銷」兩個字就是加分。

陷阱清單(oracle 會抓的違規)

  • 0 塊的單:沒有工可派,收單即完成——漏掉它的實作讓這單永遠不 submit(boundary 測試第一格)。
  • 同一塊派兩次 / 對忙碌 engine 派工 / 對已取消單派工:三種當場 panic。
  • 座標系:block_num 是邏輯序號、block_start_pos = start + b 是物理位置,oracle 驗 pair 一致。
  • submit 不完整的單 / submit 兩次 / submit 已取消的單:對帳全抓。

// ISR:三禁之下的最小形狀——搬(O(1)/筆)、計數(Relaxed 就夠:只是統計,
// 不同步任何資料)、叫醒(一波一次,合併友好)。
pub fn isr(fifo: &mut HwFifo, ring: &Ring, waker: &Waker, dropped: &AtomicU64) {
    while let Some(s) = fifo.read_fifo() {          // drain 硬體 FIFO(watermark=一波)
        if ring.try_push(s).is_err() {
            dropped.fetch_add(1, Ordering::Relaxed); // drop-newest + 可觀測
        }
    }
    waker.wake();                                   // 一波叫一次;反正會合併
}

pub fn worker_loop(ring: &Ring, waker: &Waker, stop: &AtomicBool, out: &Mutex<Vec<Sample>>) {
    loop {
        let mut batch = Vec::new();
        while let Some(s) = ring.try_pop() { batch.push(s); }   // 批次 drain
        if !batch.is_empty() { out.lock().unwrap().extend(batch); } // 一次鎖寫出
        if stop.load(Ordering::SeqCst) {
            // stop 後最後一輪 drain:「先 flag 後 wake」協定下,殘貨保證不蒸發
            let mut rest = Vec::new();
            while let Some(s) = ring.try_pop() { rest.push(s); }
            if !rest.is_empty() { out.lock().unwrap().extend(rest); }
            return;
        }
        waker.sleep();   // 醒了不代表有事——回迴圈頂重 drain 就是答案
    }
}

pub fn aggregator_loop(chs: &[Arc<Chan>], waker: &Waker, stop: &AtomicBool,
                       budget: usize, out: &Mutex<Vec<Record>>) {
    loop {
        // 中層:掃到「一整輪零收穫」為止——這一刻所有 ring 同時是空的
        loop {
            let mut moved = false;
            for ch in chs {                                // 內層:round-robin 每條 channel
                let mut batch = Vec::new();
                while batch.len() < budget {               // 公平閘:每輪每條最多 budget 筆
                    match ch.try_pop() {
                        Some(r) => batch.push(r),
                        None => break,
                    }
                }
                if !batch.is_empty() {
                    moved = true;
                    out.lock().unwrap().extend(batch);     // 批次一次鎖
                }
            }
            if !moved { break; }                          // 零收穫一整輪 → 才准考慮睡
        }
        if stop.load(Ordering::SeqCst) { return; }        // 剛 drain 到全空,殘貨已清,直接退
        waker.sleep();                                     // 貼紙補縫隙,同 j
    }
}

// driver state:兩個單調序號 + 容量,一個欄位都不多。head 活在 device 的暫存器裡。
pub fn submit(dev: &mut Device, drv: &mut Driver, tag: u32, payload: u64) -> Result<(), Full> {
    let head = dev.mmio_read(Reg::SubmitHead);
    if drv.tail - head == drv.cap {          // 滿判定:序號差,不用 % 比較
        return Err(Full);                   // 立刻回、state 一根手指不動
    }
    let idx = (drv.tail % drv.cap) as usize;   // 邏輯座標 → 物理格子
    dev.slot_write(idx, Descriptor { tag, payload });
    dev.barrier();                             // 鐵律:填完 → 柵欄 → 敲鈴
    drv.tail += 1;
    dev.mmio_write(Reg::Doorbell, drv.tail);
    Ok(())
}

pub fn poll_completions(dev: &mut Device, drv: &mut Driver, on_done: &mut dyn FnMut(u32, u64)) {
    let tail = dev.mmio_read(Reg::CompTail);   // 先讀進度(load-acquire 的硬體版)
    while drv.comp_head < tail {               // 才准碰那之前的 slot
        let idx = (drv.comp_head % drv.cap) as usize;
        let d = dev.comp_slot_read(idx);
        on_done(d.tag, d.payload);             // tag 天生路由,亂序免改碼
        drv.comp_head += 1;
    }
}

pub fn fire_timeouts(&mut self, bus: &mut impl DmaBus, now: u64) -> bool {
    let mut fired = false;
    for e in 0..ENGINE_COUNT as usize {
        if self.deadline[e].is_some_and(|d| d <= now) {
            let (rid, b) = self.owner[e].take().expect("有錶必有 owner");
            self.deadline[e] = None;
            // 注意這裡沒有 free.push_back(e) —— 隔離,不回 pool
            fired = true;
            let t = self.tries.entry((rid, b)).or_insert(0);
            *t += 1;
            if *t >= MAX_TRIES {
                if self.reqs.remove(&rid).is_some() { bus.submit_dma_request_error(rid); }
            } else {
                self.work.push_back((rid, b));   // 隔離舊台 ⇒ 同塊單一在飛,重派合法
            }
        }
    }
    fired
}

pub fn on_done(&mut self, bus: &mut impl DmaBus, eid: u32) {
    let Some((rid, _b)) = self.owner[eid as usize].take() else {
        self.free.push_back(eid);   // zombie:只當生存證明,隔離台復活;帳一根指頭不碰
        return;
    };
    self.deadline[eid as usize] = None;
    self.free.push_back(eid);
    let Some(st) = self.reqs.get_mut(&rid) else {
        return;                      // 單已 error 退場,塊的完成不再有意義
    };
    st.done += 1;
    if st.done == st.total { bus.submit_dma_request_result_done(rid); self.reqs.remove(&rid); }
}

// 醒法也是設計:睡到最近的錶;沒錶用一塊的節拍當輪詢間隔。
fn next_sleep_ms(&self, now: u64) -> u64 {
    self.deadline.iter().flatten().min()
        .map_or(BLOCK_MS, |&dl| dl.saturating_sub(now).max(1))
}

pub fn register(&mut self, j: Job) {
    let seq = self.seq; self.seq += 1;          // 到達序:到達時發,不是入場時發
    let missing = j.deps.iter()
        .filter(|d| !self.completed.contains(d)).count();   // 只數「還沒完成」的相依
    for d in &j.deps {
        if !self.completed.contains(d) {
            self.dependents.entry(*d).or_default().push(j.job_id);  // 反向邊
        }
    }
    if missing == 0 {
        self.ready.push((j.priority, Reverse(seq), j.job_id));  // 平手規則長在 tuple 裡
    } else {
        self.waiting.insert(j.job_id, Waiting { priority: j.priority, seq, missing });
    }
}

pub fn on_done(&mut self, bus: &mut impl JobBus, w: u32) {
    let id = self.owner[w as usize].take().expect("done 來自沒派工的 worker");
    self.free.push_back(w);
    bus.submit_job_done(id);
    self.completed.insert(id);
    for dep_id in self.dependents.remove(&id).unwrap_or_default() {
        let wtg = self.waiting.get_mut(&dep_id).expect("dependents 指向不存在的 waiting");
        wtg.missing -= 1;
        if wtg.missing == 0 {                   // 歸零才放行——Kahn 的事件驅動版
            let wtg = self.waiting.remove(&dep_id).unwrap();
            self.ready.push((wtg.priority, Reverse(wtg.seq), dep_id));  // 帶原到達序
        }
    }
}
trade-off:heap vs 線性掃 max

ready 用 BinaryHeap(O(log n) push/pop)vs 每次線性掃 max(O(n)): n 小時掃也行,但 heap 順手就有平手規則的位置(tuple 序),面試檯面上直接選 heap, 然後把省下的話語權花在兩層閘門的講解上。

  • 漏 completed 集合:後到者死等已逝事件(late_arrival 測試)。
  • 同權不 FIFO / seq 發錯時機:順序不穩,測試難寫、行為難重現。
